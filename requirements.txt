# SUD Workbench — runtime dependencies
# The app runs on Python 3.12 (spaCy/stanza/torch wheels are unreliable on 3.14).

# ── native shell ────────────────────────────────────────────────────────────
pywebview==6.2.1      # native macOS window, menu, dialogs, js_api bridge, http_server
                      #   pulls pyobjc-core / -Cocoa / -Quartz / -WebKit (12.2.x, known-good)
pythonnet; sys_platform == "win32"   # the Windows half of that: WinForms + WebView2 through the CLR
                      #   bridge. Already implied by pywebview's own metadata; stated here for the
                      #   same reason as in requirements-core.txt (which is what the shipping
                      #   bundles install from) — keep the two files in step. No effect on macOS:
                      #   pip evaluates the marker and skips the line entirely.
PyGObject; sys_platform == "linux"   # the Linux half: GTK3 + WebKit2GTK 4.1 through PyGObject's
pycairo; sys_platform == "linux"     #   introspection bridge. Both are pip-installable, but the GTK3
                      #   runtime + gir1.2-webkit2-4.1 themselves are NOT — those are system packages
                      #   (GObject-Introspection typelibs have no PyPI wheel path) that must already
                      #   be present; see packaging/ for the apt/dnf/pacman lines once that phase lands.
conllu>=6.0           # available for reference; I/O here is hand-rolled for byte-stable round-trips

# ── numpy ────────────────────────────────────────────────────────────────────
numpy<2; sys_platform == "darwin" and platform_machine == "x86_64"
                      # PyTorch's final macOS-x86_64 build (2.2.2 — Apple dropped Intel wheels after
                      # that) predates the numpy 2.0 C-ABI, so an unconstrained resolve on Intel Mac
                      # gives you a current (2.x) numpy alongside that old torch and every torch op
                      # raises "RuntimeError: Numpy is not available!" at runtime — not a missing
                      # dependency, an incompatible pair. This one line is what actually fixes it in
                      # THIS file: unlike the shipping bundles (requirements-core.txt + the Stanza
                      # tier's own numpy<2 pin in app/extras.py's `_stanza_platform_pins()`, installed
                      # into a separate EXTRAS_DIR appended to sys.path *after* this venv's own
                      # site-packages), everything here shares one flat site-packages, so whichever
                      # numpy pip resolves for THE WHOLE ENVIRONMENT is the one every `import numpy`
                      # gets — including inside stanza's torch. `platform_machine` (not `sys_platform`
                      # alone) is required: Windows and Linux both report "x86_64"/"AMD64" too, and
                      # this must stay a no-op there, where torch keeps shipping numpy-2-safe builds.
# ── parsing (SUD spaCy + Stanza UD) ─────────────────────────────────────────
spacy==3.8.14         # loads the sud_<lang>_<treebank> parser packages. Was 3.8.7; the bundled en_sud_ewt
                      #   wheel below requires >=3.8.14,<3.9 and pip won't resolve it against the older pin
                      #   (3.8.14 is what this project's own .venv has been running on regardless)
click>=8.0            # spaCy CLI needs it (not always pulled transitively by the resolver)
# The one model that ships WITH the app (11 MB) instead of being a Model Manager download — the
# app itself needs it: app/wiktionary.py SUD-parses each Wiktionary definition to condense it into
# a glossable phrase. Pinned to the GitHub Release asset (these packages aren't on PyPI); keep the
# version in step with requirements-core.txt, which is what both shipping bundles install from.
en_sud_ewt @ https://github.com/SunflowerAI/sud-spacy-parsers/releases/download/v0.1.0/en_sud_ewt-0.1.0-py3-none-any.whl
stanza==1.10.1        # UD parsers; pulls torch (CPU/MPS wheel on Apple Silicon)
spacy-stanza @ git+https://github.com/omri374/spacy-stanza.git   # Stanza→spaCy Doc bridge (fork relaxes the spaCy pin)

# ── Wiktionary DEFINITION lookup (diagram right-click → "Definitions of …" → MGloss pre-fill) ──
requests                      # queries Wiktionary's own REST API (definition endpoint) — no HTML scraping
beautifulsoup4                # strips markup out of the API's HTML-fragment definition text
pypinyin              # Chinese → Hànyǔ Pīnyīn / Zhuyin (context-dependent readings); also drives Gwoyeu Romatzyh
janome                # Japanese → kanji readings (pure-Python morphological analyser)
pykakasi              #   kana → phonemic Hepburn (long vowels doubled, from Janome's phonetic reading)
cutlet                # Japanese → Kunrei / Modified Hepburn romaji (per-scheme transliteration menu)
unidic-lite           #   MeCab dictionary cutlet reads for kanji readings
aksharamukha          # Sanskrit → Indic scripts (Devanagari, Grantha, Siddham, …) — per-token script conversion
indic-transliteration # the Sanskrit parser's OWN Devanagari→IAST front end. sa_sud_vedic_ufal_dcs
                      # romanises Devanagari input with it before tokenising, and its wheel declares only
                      # spaCy — so a Devanagari Sanskrit file would fail at the first parse without it.
                      # NOT interchangeable with aksharamukha above: upstream evaluated aksharamukha for
                      # this and rejected it (it renders । as '.', which breaks the model's own
                      # danda-based sentence splitting, and it is AGPL against that project's MIT).
ToJyutping            # Cantonese → Jyutping
opencc-python-reimplemented   # simplified↔traditional Han conversion (Simplified/Traditional orthographies; folds simplified input for Chao's General Chinese)
hangul-romanize       # Korean → romanisation (wiktra's Korean modules don't run on Lua 5.5)
uroman                # ISI Universal Romanizer — vocalisation-aware fallback (Persian, Hebrew, … — scripts wiktra leaves unchanged)
                      #   Arabic transliteration uses camel-tools, installed with the ar_sud_* model
